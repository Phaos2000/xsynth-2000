# Every engine/editor target uses the same immutable ROM and off-thread state helpers.
get_property(xsynth_local_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(xsynth_target IN LISTS xsynth_local_targets)
 if(NOT xsynth_target STREQUAL "xsynth_table_assets")
  get_target_property(xsynth_kind ${xsynth_target} TYPE)
  if(xsynth_kind STREQUAL "EXECUTABLE" OR xsynth_kind STREQUAL "MODULE_LIBRARY" OR xsynth_kind STREQUAL "SHARED_LIBRARY")
   target_link_libraries(${xsynth_target} PRIVATE xsynth_table_assets)
   if(MSVC AND xsynth_kind STREQUAL "EXECUTABLE")
    target_link_options(${xsynth_target} PRIVATE /STACK:8388608)
   endif()
  endif()
 endif()
endforeach()
